import { useEffect, useState } from "react";
import { useSearchParams, useNavigate } from "react-router-dom";
import { supabase } from "@/lib/supabase";
import MemberCard from "@/components/MemberCard";
import StickyBottomButton from "@/components/StickyBottomButton";
import BackButton from "@/components/BackButton";

export default function Overview() {
  const [params] = useSearchParams();
  const navigate = useNavigate();
  const groupId = params.get("groupId");

  const [members, setMembers] = useState<any[]>([]);
  const [availability, setAvailability] = useState<any[]>([]);
  const [restrictions, setRestrictions] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    loadData();
  }, []);

  async function loadData() {
    if (!groupId) {
      console.error("Missing groupId in URL");
      setLoading(false);
      return;
    }

    setLoading(true);

    const { data: membersData, error: membersError } = await supabase
      .from("members")
      .select("*")
      .eq("group_id", groupId);

    if (membersError) {
      console.error("Members error:", membersError);
      setLoading(false);
      return;
    }

    const memberIds = membersData?.map((m) => m.id) || [];

    const [availabilityRes, restrictionsRes] = await Promise.all([
      supabase.from("availability").select("*").in("member_id", memberIds),
      supabase.from("restrictions").select("*").in("member_id", memberIds),
    ]);

    setMembers(membersData || []);
    setAvailability(availabilityRes.data || []);
    setRestrictions(restrictionsRes.data || []);
    setLoading(false);
  }

  function getMemberAvailability(id: string) {
    return availability
      .filter((a) => a.member_id === id)
      .map((a) => a.time_slot);
  }

  function getMemberRestrictions(id: string) {
    return restrictions
      .filter((r) => r.member_id === id)
      .map((r) => r.restriction);
  }

  function handleContinue() {
    navigate(`/browse?groupId=${groupId}`);
  }

  if (loading) {
    return (
      <div className="min-h-screen flex items-center justify-center bg-background">
        <div className="text-sm text-muted-foreground">
          Loading group overview…
        </div>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-background px-6 pt-24 pb-24">

      <div className="max-w-2xl mx-auto pb-32">

        <BackButton />

        <h1 className="text-3xl font-bold tracking-tight text-foreground">
          Group Overview
        </h1>

        <p className="text-sm text-muted-foreground mt-1">
          A quick look at everyone’s availability and dietary needs.
        </p>

        <div className="space-y-4 mt-6">
          {members.length === 0 ? (
            <p className="text-muted-foreground text-sm">
              No members found for this group.
            </p>
          ) : (
            members.map((m) => (
              <MemberCard
                key={m.id}
                member={m}
                availability={getMemberAvailability(m.id)}
                restrictions={getMemberRestrictions(m.id)}
              />
            ))
          )}
        </div>
      </div>

      <StickyBottomButton onClick={handleContinue}>
        Continue to Restaurants
      </StickyBottomButton>
    </div>
  );
}
