import { useState } from "react";
import { useSearchParams, useNavigate } from "react-router-dom";
import { supabase } from "@/lib/supabase";
import BackButton from "@/components/BackButton";
import ProgressStep from "@/components/ProgressStep";

export default function DietaryRestrictions() {
  const [params] = useSearchParams();
  const navigate = useNavigate();

  const memberId = params.get("memberId");
  const groupId = params.get("groupId");

  const options = [
    "Vegetarian",
    "Vegan",
    "Gluten-free",
    "Lactose-free",
    "Nut allergy",
    "Shellfish allergy",
    "No pork",
    "No beef",
  ];

  const [selected, setSelected] = useState<string[]>([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");

  function toggle(item: string) {
    setSelected((prev) =>
      prev.includes(item)
        ? prev.filter((x) => x !== item)
        : [...prev, item]
    );
  }

  async function handleSubmit() {
    if (!memberId) {
      setError("Missing member ID");
      return;
    }

    setLoading(true);
    setError("");

    const rows = selected.map((item) => ({
      member_id: memberId,
      restriction: item,
    }));

    const { error } = await supabase.from("restrictions").insert(rows);

    setLoading(false);

    if (error) {
      setError(error.message);
      return;
    }

    navigate(`/overview?groupId=${groupId}&memberId=${memberId}`);
  }

  function handleSkip() {
    navigate(`/overview?groupId=${groupId}&memberId=${memberId}`);
  }

  return (
    <div className="min-h-screen flex items-center justify-center bg-background p-6">
      <div className="w-full max-w-md bg-card shadow-lg rounded-xl p-6 border border-border">

        <BackButton />

        <ProgressStep step={3} total={4} />

        <h1 className="text-2xl font-semibold mb-4 text-foreground">
          Dietary Restrictions
        </h1>

        <p className="text-sm text-muted-foreground mb-4">
          Select any dietary restrictions you have.
        </p>

        <div className="grid grid-cols-2 gap-3 mb-6">
          {options.map((item) => {
            const isSelected = selected.includes(item);
            return (
              <button
                key={item}
                onClick={() => toggle(item)}
                className={`
                  py-2 px-3 rounded-md text-sm font-medium tracking-tight border border-border transition
                  ${
                    isSelected
                      ? "bg-[hsl(var(--button-primary))] text-[hsl(var(--button-primary-foreground))]"
                      : "bg-background text-foreground hover:bg-muted"
                  }
                `}
              >
                {item}
              </button>
            );
          })}
        </div>

        {error && (
          <p className="text-red-500 text-sm mb-3">{error}</p>
        )}

        <button
          onClick={handleSubmit}
          disabled={loading}
          className="
            w-full py-2 rounded-md text-sm font-medium tracking-tight disabled:opacity-50 mb-3
            bg-[hsl(var(--button-primary))]
            text-[hsl(var(--button-primary-foreground))]
          "
        >
          {loading ? "Saving..." : "Continue"}
        </button>

        <button
          onClick={handleSkip}
          className="w-full text-sm text-muted-foreground underline hover:text-foreground transition"
        >
          Skip for now
        </button>
      </div>
    </div>
  );
}
