import { Clock, Shield } from "lucide-react";

interface MemberCardProps {
  member: {
    id: string;
    name: string;
    is_organizer?: boolean;
  };
  availability: string[];
  restrictions: string[];
}

const MemberCard = ({ member, availability, restrictions }: MemberCardProps) => {
  return (
    <div className="bg-card border border-border rounded-sm p-4 space-y-3">

      {/* Header */}
      <div className="flex items-center justify-between">
        <div className="flex items-center gap-2">

          {/* Square avatar */}
          <div className="h-8 w-8 rounded-sm bg-muted flex items-center justify-center text-xs font-medium text-foreground">
            {member.name?.[0]?.toUpperCase() || "?"}
          </div>

          <h3 className="font-semibold text-foreground text-base">
            {member.name}
          </h3>
        </div>

        {member.is_organizer && (
          <span className="inline-flex items-center gap-1 bg-primary/10 text-primary text-[10px] px-2 py-1 rounded-sm font-semibold uppercase tracking-wide">
            <Shield className="w-3 h-3" />
            Organizer
          </span>
        )}
      </div>

      {/* Availability */}
      {availability.length > 0 && (
        <div className="flex items-start gap-2">
          <Clock className="w-3.5 h-3.5 text-muted-foreground mt-0.5 shrink-0" />
          <p className="text-sm text-muted-foreground leading-relaxed">
            {availability.join(", ")}
          </p>
        </div>
      )}

      {/* Restrictions */}
      {restrictions.length > 0 && (
        <div className="flex flex-wrap gap-1.5">
          {restrictions.map((r) => (
            <span
              key={r}
              className="inline-flex items-center bg-muted text-foreground px-2 py-0.5 text-[11px] rounded-sm font-medium border border-border"
            >
              {r}
            </span>
          ))}
        </div>
      )}

      {/* Empty state */}
      {availability.length === 0 && restrictions.length === 0 && (
        <p className="text-sm text-muted-foreground">No details added yet</p>
      )}
    </div>
  );
};

export default MemberCard;
